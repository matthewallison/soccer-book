  \begin{tikzpicture}[remember picture, overlay]
    \fill[hbgreen] (current page.south west) rectangle (current page.north east);
    \draw[white, opacity=0.13, line width=1.6pt] ($(current page.center)+(0,-1.2in)$) circle (1.9in);
    \draw[white, opacity=0.13, line width=1.6pt] ($(current page.west)+(0,-1.2in)$) -- ($(current page.east)+(0,-1.2in)$);
    \fill[white, opacity=0.13] ($(current page.center)+(0,-1.2in)$) circle (4pt);
  \end{tikzpicture}%
  \vspace*{1.6in}%
  {\centering
    {\sffamily\bfseries\small\color{white!80!hbgreen}\addfontfeatures{LetterSpace=18}\MakeUppercase{#2}\par}
    \vspace{10pt}
    {\hbblack\fontsize{40}{46}\selectfont\color{white}#3\par}
    \vspace{14pt}
    {\hblight\fontsize{14}{19}\selectfont\color{white!88!hbgreen}\parbox{4.4in}{\centering #4}\par}
  }%
  \clearpage
}
