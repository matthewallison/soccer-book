% ---------- Page breaks ----------
% \raggedbottom only adds fill at the page bottom when a page is shipped out.
% While choosing a break TeX sees almost no stretch, so every short page scores
% as equally bad and the heading rules below would win even with inches left.
% Stretch in \topskip (plain TeX's ragged-bottom value) lets TeX weigh how
% short a page is against the penalties; the bottom fill keeps pages ragged.
\setlength{\topskip}{11pt plus 60pt}
\makeatletter
% Never leave a single line of a paragraph or list item alone at the top or
% bottom of a page.

% Used by handbook.lua inside list items: \hbkeepnextitem forbids a break
% before the next \item and \hbreleaseitem restores the normal penalty (so no
% list leaves its first or last item alone); \hblistendpenalty sets the
% penalty for a break after the list (a cue box that closes it).
\def\hb@itempenalty{-\@lowpenalty}

\newcommand{\hbreleaseitem}{\@itempenalty=\hb@itempenalty\relax}
\newcommand{\hblistendpenalty}[1]{\@endparpenalty=#1\relax}

    \fill[hbgreen] (current page.south west) rectangle (current page.north east);
    \draw[white, opacity=0.13, line width=1.6pt] ($(current page.center)+(0,-1.2in)$) circle (1.9in);
    \draw[white, opacity=0.13, line width=1.6pt] ($(current page.west)+(0,-1.2in)$) -- ($(current page.east)+(0,-1.2in)$);
    \fill[white, opacity=0.13] ($(current page.center)+(0,-1.2in)$) circle (4pt);

  \vspace*{1.6in}%
  {\centering
    {\sffamily\bfseries\small\color{white!80!hbgreen}\addfontfeatures{LetterSpace=18}\MakeUppercase{#2}\par}
    \vspace{10pt}
    {\hbblack\fontsize{40}{46}\selectfont\color{white}#3\par}
    \vspace{14pt}
    {\hblight\fontsize{14}{19}\selectfont\color{white!88!hbgreen}\parbox{5.2in}{\centering #4}\par}
  }%
